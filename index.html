// ========================================== // 1. ประตูรับข้อมูลเพื่อบันทึก ลบ ล็อกอิน และจัดการผู้ใช้ (POST Request) // ========================================== function doPost(e) { // สร้างระบบ Lock เพื่อป้องกันการรันพร้อมกัน (Race Condition) var lock = LockService.getScriptLock(); // 1. พยายามขอคิวเข้าทำงาน รอสูงสุด 15 วินาที try { lock.waitLock(15000); } catch (lockError) { // ถ้าเกิน 15 วินาทีแล้วยังมีคนบันทึกอยู่ ให้แจ้งเตือนกลับไป return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "ระบบมีผู้ใช้งานหนาแน่น กำลังบันทึกข้อมูลพร้อมกัน กรุณากดปุ่มบันทึกใหม่อีกครั้ง" })).setMimeType(ContentService.MimeType.JSON); } // 2. เมื่อได้คิวแล้ว เริ่มทำงาน try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Sheet1"); // ชีตเก็บข้อมูลเรือ var sheetUsers = ss.getSheetByName("Users"); // ชีตเก็บข้อมูลผู้ใช้งาน // ตรวจสอบว่ามีชีต Sheet1 หรือไม่ if (!sheet) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "ไม่พบแท็บข้อมูลชื่อ Sheet1" })) .setMimeType(ContentService.MimeType.JSON); } // ถ้ายังไม่มีชีต Users ให้สร้างอัตโนมัติและใส่ Admin เริ่มต้น if (!sheetUsers) { sheetUsers = ss.insertSheet("Users"); sheetUsers.appendRow(["Username", "Password", "Role"]); sheetUsers.appendRow(["admin", "123456", "admin"]); // แอดมินเริ่มต้น } var action = data.action; // ---------------------------------------------------- // คำสั่ง: เข้าสู่ระบบ (Login) // ---------------------------------------------------- if (action === "login") { var usersData = sheetUsers.getDataRange().getValues(); for (var i = 1; i < usersData.length; i++) { if (usersData[i][0] == data.username && usersData[i][1] == data.password) { return ContentService.createTextOutput(JSON.stringify({ "result": "success", "username": usersData[i][0], "role": usersData[i][2] // ส่งสิทธิ์กลับไป (admin หรือ user) })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง" })) .setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // คำสั่ง: ลบข้อมูลเรือ (Delete) // ---------------------------------------------------- if (action === "delete") { var targetId = data.boatId; var dbData = sheet.getDataRange().getValues(); var deleted = false; for (var j = 1; j < dbData.length; j++) { if (dbData[j][0].toString().trim() === targetId) { sheet.deleteRow(j + 1); // ลบแถว deleted = true; break; } } if(deleted) { return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "ไม่พบข้อมูลรหัสเรือที่ต้องการลบ" })).setMimeType(ContentService.MimeType.JSON); } } // ---------------------------------------------------- // คำสั่ง: สร้างผู้ใช้ใหม่ (Create User) // ---------------------------------------------------- if (action === "createUser") { var existingUsers = sheetUsers.getDataRange().getValues(); for (var k = 1; k < existingUsers.length; k++) { if (existingUsers[k][0] == data.newUsername) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "มีชื่อผู้ใช้นี้อยู่แล้วในระบบ" })) .setMimeType(ContentService.MimeType.JSON); } } sheetUsers.appendRow([data.newUsername, data.newPassword, data.role]); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // คำสั่ง: บันทึกข้อมูลเรือใหม่ (Insert) // ---------------------------------------------------- if (action === "insert" || !action) { var prefix = ""; if (data.boatId && data.boatId.indexOf("-") > -1) { var parts = data.boatId.split("-"); parts.pop(); prefix = parts.join("-"); } else { prefix = (data.moo || "-") + " " + (data.district === "โพธิ์ไทร" ? "พ." : "ดต."); } var lastRow = sheet.getLastRow(); var nextNumber = 1; if (lastRow > 1) { var lastBoatId = sheet.getRange(lastRow, 1).getValue().toString(); var lastParts = lastBoatId.split('-'); if (lastParts.length >= 2) { var lastNum = parseInt(lastParts[lastParts.length - 1], 10); if (!isNaN(lastNum)) { nextNumber = lastNum + 1; } else { nextNumber = lastRow; } } else { nextNumber = lastRow; } } var formattedNumber = ("000" + nextNumber).slice(-3); var finalBoatId = prefix + "-" + formattedNumber; var boatType = "ไม่ระบุ (ดึงจากฟอร์มเดิมไม่ได้ส่งมา)"; sheet.appendRow([ finalBoatId, boatType, data.ownerName, data.idcard, data.addressNo, data.moo, data.subDistrict, data.district, data.province, data.phoneNumber, data.mooringPoint, data.ownerImage, data.boatImage ]); return ContentService.createTextOutput(JSON.stringify({ "result": "success", "savedBoatId": finalBoatId })).setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // คำสั่ง: อัปเดตข้อมูลเรือเดิม (Update) // ---------------------------------------------------- if (action === "update") { var targetId = data.boatId; var dbData = sheet.getDataRange().getValues(); var rowIndex = -1; for (var r = 1; r < dbData.length; r++) { if (dbData[r][0].toString().trim() === targetId) { rowIndex = r + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 3, 1, 11).setValues([[ data.ownerName, data.idcard, data.addressNo, data.moo, data.subDistrict, data.district, data.province, data.phoneNumber, data.mooringPoint, data.ownerImage, data.boatImage ]]); return ContentService.createTextOutput(JSON.stringify({ "result": "success", "savedBoatId": targetId })).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "ไม่พบข้อมูลที่ต้องการแก้ไข" })).setMimeType(ContentService.MimeType.JSON); } } // ^^^ ปิดปีกกาของ update เรียบร้อยตรงนี้ ^^^ // ---------------------------------------------------- // คำสั่ง: ดึงข้อมูลทั้งหมดสำหรับสร้างรายงาน Excel (แยกออกมาอิสระแล้ว) // ---------------------------------------------------- if (action === "exportData") { var dbData = sheet.getDataRange().getValues(); var exportResult = []; // ลูปดึงข้อมูล เริ่มจาก i = 1 เพื่อข้ามหัวตาราง (Header) for (var i = 1; i < dbData.length; i++) { // เช็คว่าแถวนั้นมีรหัสเรืออยู่จริงๆ ไม่ใช่แถวว่าง if (dbData[i][0] && dbData[i][0].toString().trim() !== "") { exportResult.push({ boatId: dbData[i][0], boatType: dbData[i][1], ownerName: dbData[i][2], idcard: dbData[i][3], addressNo: dbData[i][4], moo: dbData[i][5], subDistrict: dbData[i][6], district: dbData[i][7], province: dbData[i][8], phoneNumber: dbData[i][9], mooringPoint: dbData[i][10] }); } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "data": exportResult })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "Invalid action parameter" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // ========================================== // 2. ประตูจัดการหน้าเว็บและการค้นหา (GET Request) // ========================================== function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Sheet1"); // ดึงเลขล่าสุดเพื่อเอาไปแสดงผลชั่วคราวบนหน้าเว็บ if (e.parameter.getLatestCount) { var lastRow = sheet.getLastRow(); var nextNumber = 1; if (lastRow > 1) { var lastBoatId = sheet.getRange(lastRow, 1).getValue().toString(); var lastParts = lastBoatId.split('-'); if (lastParts.length >= 2) { var lastNum = parseInt(lastParts[lastParts.length - 1], 10); if (!isNaN(lastNum)) { nextNumber = lastNum + 1; } } else { nextNumber = lastRow; } } return ContentService.createTextOutput(JSON.stringify({ "nextNumber": nextNumber })) .setMimeType(ContentService.MimeType.JSON); } // ค้นหาข้อมูลเรือ if (e.parameter.searchId && e.parameter.fetchData === "true") { var searchId = e.parameter.searchId.trim(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() === searchId) { return ContentService.createTextOutput(JSON.stringify({ "result": "success", "found": true, "boatId": data[i][0], "boatType": data[i][1], "ownerName": data[i][2], "idcard": data[i][3], "addressNo": data[i][4], "moo": data[i][5], "subDistrict": data[i][6], "district": data[i][7], "province": data[i][8], "phoneNumber": data[i][9], "mooringPoint": data[i][10], "ownerImage": data[i][11], "boatImage": data[i][12] })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ "result": "success", "found": false })) .setMimeType(ContentService.MimeType.JSON); } // โหลดหน้าเว็บ HTML var template = HtmlService.createTemplateFromFile('index'); return template.evaluate() .setTitle('ระบบจัดระเบียบเรืออำเภอดอนตาล') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } catch (error) { return HtmlService.createHtmlOutput("Error: " + error.toString()); } }
